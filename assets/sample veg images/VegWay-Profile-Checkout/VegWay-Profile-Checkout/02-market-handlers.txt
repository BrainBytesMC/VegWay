// REPLACE the existing ProfileAction definition; do not append a duplicate.
void VegWayMarket::ProfileAction(System::Object^ sender, System::EventArgs^)
{
    if (IsDesignSurface()) return;
    if (System::String::IsNullOrWhiteSpace(userId)) return;
    System::Windows::Forms::ToolStripMenuItem^ item =
        safe_cast<System::Windows::Forms::ToolStripMenuItem^>(sender);
    int page = 0;
    if (item == changeNameItem) page = 1;
    else if (item == changePasswordItem) page = 2;
    else if (item == toolStripMenuItem1) page = 3;
    VegWay::ProfileSettings^ settings = gcnew VegWay::ProfileSettings(userId, page);
    try {
        settings->ShowDialog(this);
        if (!System::String::IsNullOrWhiteSpace(settings->UpdatedName))
            userName = settings->UpdatedName;
    }
    finally { delete settings; }
    LoadProducts();
}

// REPLACE the existing BuyClick definition; do not append a duplicate.
void VegWayMarket::BuyClick(System::Object^ sender, System::EventArgs^)
{
    if (IsDesignSurface()) return;
    if (System::String::IsNullOrWhiteSpace(userId)) {
        System::Windows::Forms::MessageBox::Show(this, "Please log in before ordering.");
        return;
    }
    System::String^ itemId = safe_cast<System::String^>(
        safe_cast<System::Windows::Forms::Button^>(sender)->Tag);
    // The attached PayNow uses LOWERCASE namespace vegway. Preserve it.
    vegway::PayNow^ checkout = gcnew vegway::PayNow(userId, itemId);
    try { checkout->ShowDialog(this); }
    finally { delete checkout; }
    // Refresh even after cancelling, since other buyers may have changed stock.
    LoadProducts();
}
